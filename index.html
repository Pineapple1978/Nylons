<script>


function calculate(){


let material=document.getElementById("material").value;


let ldpe=parseFloat(document.getElementById("ldpeDensity").value);


let hdpe=parseFloat(document.getElementById("hdpeDensity").value);


let width=parseFloat(document.getElementById("width").value);


let length=parseFloat(document.getElementById("length").value);


let micron=parseFloat(document.getElementById("thickness").value);


let priceKg=parseFloat(document.getElementById("priceKg").value);


// Convert micron to cm

let thicknessCM=micron/10000;


// Volume

let volume=width*length*thicknessCM;


// Density according to selected material

let density;


if(material==="LDPE"){

    density=ldpe;

}else{

    density=hdpe;

}


// Weight

let weight=volume*density;


// Price

let price=(weight/1000)*priceKg;


// Display

document.getElementById("density").innerHTML=density.toFixed(2);


document.getElementById("volume").innerHTML=volume.toFixed(2);


document.getElementById("weight").innerHTML=weight.toFixed(2);


document.getElementById("price").innerHTML=price.toFixed(2);


}


// Auto Calculate

document.querySelectorAll("input,select").forEach(function(item){


item.addEventListener("input",calculate);


item.addEventListener("change",calculate);


});


calculate();


</script>
